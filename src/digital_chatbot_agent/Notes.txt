from langchain_core.documents import Document
from langchain_chroma import Chroma
from langchain_huggingface import HuggingFaceEmbeddings

embeddings = HuggingFaceEmbeddings(model_name="sentence-transformers/all-MiniLM-L6-v2")

# Our chunks as Documents (text + metadata)
chunks = [
    Document(page_content="Refunds are available within 14 days of purchase.",
             metadata={"source": "policies.txt"}),
    Document(page_content="The Growth package costs $999 per month and includes ads management.",
             metadata={"source": "packages.txt"}),
    Document(page_content="You can cancel any package with 30 days written notice.",
             metadata={"source": "policies.txt"}),
]

# Create the store: embeds every chunk and saves it to the ./chroma_db folder
vector_store = Chroma.from_documents(
    documents=chunks,                  # what to store
    embedding=embeddings,              # how to turn text into vectors
    persist_directory="./chroma_db",   # folder on disk where it is saved
)

result=vector_store.similarity_search("How can i get my money back",k=2
)
for r in result:
    print(r.page_content, "|", r.metadata)
